kafka prac9:

create yml
version: '3.8'
services:
  zookeeper:
    image: confluentinc/cp-zookeeper:7.6.0
    container_name: zookeeper
    environment:
      ZOOKEEPER_CLIENT_PORT: 2181
      ZOOKEEPER_TICK_TIME: 2000
    ports:
      - "2181:2181"
  kafka:
    image: confluentinc/cp-kafka:7.6.0
    container_name: kafka
    depends_on:
      - zookeeper
    ports:
      - "9092:9092"
    environment:
      KAFKA_BROKER_ID: 1
      KAFKA_ZOOKEEPER_CONNECT: zookeeper:2181
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://localhost:9092
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1


run yml:
docker compose up -d


create topic:

docker exec -it kakfka kafka-topics --create --topic hello --bootstrap-server localhost:9092 --partitions 1 --replication-factor 1 .

create producer ( on a different cmd):

docker exec -it kafka kafka-console-producer --topic hello --bootstrap-server localhost:9092

create consumer ( on a different cmd):

docker exec kafka kafka-console-consumer --topic hello -bootstrap-server localhost:9092
